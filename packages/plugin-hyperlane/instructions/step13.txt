# Step 13: Testing
-------------------------
Purpose: Implement pure functions for testing with Hyperlane-specific considerations

Design Decisions:
1. Unit Testing:
   - Pure test functions
   - Immutable test state
   - Composable test chain
   Justification: Test units functionally

2. Integration Testing:
   - Pure integration functions
   - Immutable integration state
   - Composable integration chain
   Justification: Test integration functionally

3. E2E Testing:
   - Pure E2E functions
   - Immutable E2E state
   - Composable E2E chain
   Justification: Test E2E functionally

Actions:
1. Pure Test Types
   ```typescript
   export type TestCase = Readonly<{
     name: string;
     input: unknown;
     expected: unknown;
     metadata?: ReadonlyMap<string, unknown>;
   }>;

   export type TestSuite = Readonly<{
     name: string;
     cases: ReadonlyArray<TestCase>;
     setup?: () => Promise<void>;
     teardown?: () => Promise<void>;
     metadata: ReadonlyMap<string, unknown>;
   }>;

   export type TestResult = Readonly<{
     name: string;
     passed: boolean;
     error?: Error;
     duration: number;
     metadata: ReadonlyMap<string, unknown>;
   }>;
   ```

2. Pure Test Functions
   ```typescript
   export const createTestRunner = (
     config: Readonly<TestConfig>
   ) =>
     async (
       suite: Readonly<TestSuite>
     ): Promise<ReadonlyArray<TestResult>> => {
       try {
         // Run setup if provided
         if (suite.setup) {
           await suite.setup();
         }
         
         // Run test cases
         const results = await Promise.all(
           suite.cases.map(async test => {
             const start = Date.now();
             
             try {
               const result = await runTest(test);
               const duration = Date.now() - start;
               
               return {
                 name: test.name,
                 passed: true,
                 duration,
                 metadata: new Map([
                   ['test', test],
                   ['result', result]
                 ])
               } as const;
             } catch (error) {
               const duration = Date.now() - start;
               
               return {
                 name: test.name,
                 passed: false,
                 error,
                 duration,
                 metadata: new Map([
                   ['test', test],
                   ['error', error]
                 ])
               } as const;
             }
           })
         );
         
         // Run teardown if provided
         if (suite.teardown) {
           await suite.teardown();
         }
         
         return results;
       } catch (error) {
         return [{
           name: suite.name,
           passed: false,
           error,
           duration: 0,
           metadata: new Map([
             ['suite', suite],
             ['error', error]
           ])
         }] as const;
       }
     };

   export const createIntegrationRunner = (
     config: Readonly<TestConfig>
   ) =>
     async (
       suites: ReadonlyArray<TestSuite>
     ): Promise<ReadonlyArray<TestResult>> => {
       try {
         // Run suites in sequence
         const results: TestResult[] = [];
         
         for (const suite of suites) {
           const runner = createTestRunner(config);
           const suiteResults = await runner(suite);
           results.push(...suiteResults);
         }
         
         return results;
       } catch (error) {
         return [{
           name: 'integration',
           passed: false,
           error,
           duration: 0,
           metadata: new Map([
             ['suites', suites],
             ['error', error]
           ])
         }] as const;
       }
     };
   ```

3. Pure E2E Functions
   ```typescript
   export const createE2ERunner = (
     config: Readonly<TestConfig>
   ) =>
     async (
       scenario: Readonly<E2EScenario>
     ): Promise<Readonly<TestResult>> => {
       const start = Date.now();
       
       try {
         // Setup environment
         await setupE2EEnvironment(scenario);
         
         // Run scenario steps
         for (const step of scenario.steps) {
           await runE2EStep(step);
         }
         
         // Verify expectations
         await verifyE2EExpectations(scenario);
         
         const duration = Date.now() - start;
         
         return {
           name: scenario.name,
           passed: true,
           duration,
           metadata: new Map([
             ['scenario', scenario],
             ['config', config]
           ])
         } as const;
       } catch (error) {
         const duration = Date.now() - start;
         
         return {
           name: scenario.name,
           passed: false,
           error,
           duration,
           metadata: new Map([
             ['scenario', scenario],
             ['error', error]
           ])
         } as const;
       }
     };

   export const createTestProcess = (
     config: Readonly<TestConfig>
   ) =>
     async (
       context: Readonly<TestContext>
     ): Promise<Readonly<Result<void>>> => {
       try {
         // Run unit tests
         const unitResults = await createTestRunner(config)(
           context.unitSuites
         );
         
         // Run integration tests
         const integrationResults = await createIntegrationRunner(config)(
           context.integrationSuites
         );
         
         // Run E2E tests
         const e2eResults = await Promise.all(
           context.e2eScenarios.map(scenario =>
             createE2ERunner(config)(scenario)
           )
         );
         
         // Check for failures
         const failures = [
           ...unitResults,
           ...integrationResults,
           ...e2eResults
         ].filter(r => !r.passed);
         
         if (failures.length > 0) {
           throw new Error(
             `Tests failed: ${failures.map(f => f.name).join(', ')}`
           );
         }
         
         return {
           success: true,
           metadata: new Map([
             ['context', context],
             ['results', {
               unit: unitResults,
               integration: integrationResults,
               e2e: e2eResults
             }]
           ])
         } as const;
       } catch (error) {
         return {
           success: false,
           error: {
             code: 'TEST_PROCESS_FAILED',
             message: error.message,
             context: new Map([['context', context]])
           }
         } as const;
       }
     };
   ```
