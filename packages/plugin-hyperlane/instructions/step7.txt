# Step 7: Message Processing & Validation
-------------------------
Purpose: Implement pure functions for message processing and validation

Documentation References:
- ElizaOS Message Processing: https://elizaos.github.io/eliza/docs/messaging/processing
- Hyperlane Message Verification: https://docs.hyperlane.xyz/docs/reference/messaging/receive
- Hyperlane Security Modules: https://docs.hyperlane.xyz/docs/reference/ISM/specify-your-ISM

Design Decisions:
1. Message Processing:
   - Pure processing functions
   - Immutable message state
   - Composable processor chain
   Justification: Maintain functional purity in message processing

2. Message Validation:
   - Pure validation functions
   - Immutable validation state
   - Composable validator chain
   Justification: Ensure message integrity with pure functions

3. Error Handling:
   - Pure error handlers
   - Immutable error state
   - Composable error chain
   Justification: Handle errors functionally and safely

Actions:
1. Pure Processing Types
   ```typescript
   export type ProcessConfig = Readonly<{
     validators: ReadonlyArray<string>;
     processors: ReadonlyArray<string>;
     errorHandlers: ReadonlyArray<string>;
     metadata?: ReadonlyMap<string, unknown>;
   }>;

   export type ValidationResult = Readonly<{
     valid: boolean;
     errors: ReadonlyArray<string>;
     metadata: ReadonlyMap<string, unknown>;
   }>;

   export type ProcessResult = Readonly<{
     messageId: string;
     status: 'processed' | 'failed';
     errors: ReadonlyArray<string>;
     metadata: ReadonlyMap<string, unknown>;
   }>;
   ```

2. Pure Processing Functions
   ```typescript
   export const createMessageProcessor = (
     config: Readonly<ProcessConfig>
   ) =>
     async (
       message: Readonly<MessageHandler>
     ): Promise<Readonly<ProcessResult>> => {
       // Validate message
       const validateMessage = createMessageValidator(config.validators);
       const validationResult = await validateMessage(message);

       if (!validationResult.valid) {
         return {
           messageId: message.id,
           status: 'failed',
           errors: validationResult.errors,
           metadata: new Map([
             ['timestamp', Date.now()],
             ['stage', 'validation']
           ])
         } as const;
       }

       try {
         // Process message through processor chain
         for (const processor of config.processors) {
           const processMessage = createProcessor(processor);
           await processMessage(message);
         }

         return {
           messageId: message.id,
           status: 'processed',
           errors: [],
           metadata: new Map([
             ['timestamp', Date.now()],
             ['processors', config.processors]
           ])
         } as const;
       } catch (error) {
         // Handle errors through error chain
         const handleError = createErrorHandler(config.errorHandlers);
         const handledError = await handleError(error, message);

         return {
           messageId: message.id,
           status: 'failed',
           errors: [handledError.message],
           metadata: new Map([
             ['timestamp', Date.now()],
             ['stage', 'processing'],
             ['error', handledError]
           ])
         } as const;
       }
     };

   export const createMessageValidator = (
     validators: ReadonlyArray<string>
   ) =>
     async (
       message: Readonly<MessageHandler>
     ): Promise<Readonly<ValidationResult>> => {
       const errors: string[] = [];
       const metadata = new Map<string, unknown>();

       for (const validator of validators) {
         const validate = createValidator(validator);
         const result = await validate(message);

         if (!result.valid) {
           errors.push(...result.errors);
         }

         // Merge metadata
         for (const [key, value] of result.metadata) {
           metadata.set(key, value);
         }
       }

       return {
         valid: errors.length === 0,
         errors: errors as ReadonlyArray<string>,
         metadata: metadata as ReadonlyMap<string, unknown>
       } as const;
     };
   ```

3. Pure Error Functions
   ```typescript
   export const createErrorHandler = (
     handlers: ReadonlyArray<string>
   ) =>
     async (
       error: Error,
       message: Readonly<MessageHandler>
     ): Promise<Readonly<ProcessedError>> => {
       let processedError = error;
       const metadata = new Map<string, unknown>();

       for (const handler of handlers) {
         const handleError = createHandler(handler);
         const result = await handleError(processedError, message);
         processedError = result.error;

         // Merge metadata
         for (const [key, value] of result.metadata) {
           metadata.set(key, value);
         }
       }

       return {
         error: processedError,
         metadata: metadata as ReadonlyMap<string, unknown>,
         timestamp: Date.now()
       } as const;
     };
   ```

5. Test Structure
   ```typescript
   // Pure processor tests
   describe('createMessageProcessor', () => {
     it('processes valid messages', async () => {
       const processor = createMessageProcessor({
         validators: ['signature', 'nonce'],
         processors: ['token', 'data'],
         errorHandlers: ['retry', 'log']
       });

       const result = await processor({
         id: '0x...',
         origin: 1,
         sender: '0x...',
         message: '0x...'
       });
       
       expect(result.status).toBe('processed');
       expect(result.errors).toHaveLength(0);
     });

     it('handles invalid messages', async () => {
       const processor = createMessageProcessor({
         validators: ['signature'],
         processors: ['token'],
         errorHandlers: ['log']
       });

       const result = await processor({
         id: '0x...',
         origin: 1,
         sender: 'invalid',
         message: '0x...'
       });
       
       expect(result.status).toBe('failed');
       expect(result.errors).toHaveLength(1);
     });
   });

   // Pure validator tests
   describe('createMessageValidator', () => {
     it('validates correct messages', async () => {
       const validator = createMessageValidator(['signature', 'nonce']);
       const result = await validator({
         id: '0x...',
         origin: 1,
         sender: '0x...',
         message: '0x...'
       });
       
       expect(result.valid).toBe(true);
       expect(result.errors).toHaveLength(0);
     });
   });

   // Pure error handler tests
   describe('createErrorHandler', () => {
     it('processes errors through chain', async () => {
       const handler = createErrorHandler(['retry', 'log']);
       const result = await handler(
         new Error('Test error'),
         {
           id: '0x...',
           origin: 1,
           sender: '0x...',
           message: '0x...'
         }
       );
       
       expect(result.error).toBeDefined();
       expect(result.metadata.get('retried')).toBe(true);
     });
   });
   ```
