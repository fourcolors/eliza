# Step 10: Gas Management
-------------------------
Purpose: Implement pure functions for cross-chain gas management

Documentation References:
- ElizaOS Gas Management: https://elizaos.github.io/eliza/docs/gas/
- Hyperlane IGP: https://docs.hyperlane.xyz/docs/reference/hooks/igp
- Hyperlane Gas Payment: https://docs.hyperlane.xyz/docs/reference/messaging/send#quote-dispatch

Design Decisions:
1. Gas Estimation:
   - Pure estimation functions
   - Immutable gas state
   - Composable estimator chain
   Justification: Maintain functional purity in gas estimation

2. Gas Payment:
   - Pure payment functions
   - Immutable payment state
   - Composable payment chain
   Justification: Handle gas payments functionally

3. Gas Optimization:
   - Pure optimization functions
   - Immutable optimization state
   - Composable optimizer chain
   Justification: Optimize gas usage functionally

2. Pure Gas Functions
   ```typescript
   export const createGasEstimator = (
     config: Readonly<GasConfig>
   ) =>
     async (
       message: Readonly<MessageHandler>
     ): Promise<Readonly<GasEstimate>> => {
       // Get appropriate estimator
       const estimator = createEstimator(config.estimator);
       
       try {
         // Get base estimate
         const base = await estimator.estimate(message);
         
         // Apply multiplier and buffer
         const gasLimit = applyMultiplier(base.gasLimit, config.multiplier);
         const totalCost = applyBuffer(base.totalCost, config.buffer);
         
         return {
           gasLimit,
           gasPrice: base.gasPrice,
           totalCost,
           metadata: new Map([
             ['config', config],
             ['base', base],
             ['timestamp', Date.now()]
           ])
         } as const;
       } catch (error) {
         throw new Error(`Gas estimation failed: ${error.message}`);
       }
     };

   export const createEstimator = (
     type: string
   ): Readonly<IGasEstimator> => {
     switch (type) {
       case 'oracle':
         return {
           estimate: createOracleEstimator(),
           metadata: new Map([
             ['type', 'oracle'],
             ['version', '1.0.0']
           ])
         } as const;
         
       case 'static':
         return {
           estimate: createStaticEstimator(),
           metadata: new Map([
             ['type', 'static'],
             ['version', '1.0.0']
           ])
         } as const;
         
       default:
         throw new Error(`Unsupported estimator type: ${type}`);
     }
   };
   ```

3. Pure Payment Functions
   ```typescript
   export const createGasPayment = (
     config: Readonly<GasConfig>
   ) =>
     async (
       estimate: Readonly<GasEstimate>
     ): Promise<Readonly<PaymentResult>> => {
       try {
         // Get payment provider
         const provider = createPaymentProvider();
         
         // Execute payment
         const tx = await provider.pay({
           value: estimate.totalCost,
           gasLimit: estimate.gasLimit
         });
         
         return {
           success: true,
           txHash: tx.hash,
           metadata: new Map([
             ['estimate', estimate],
             ['timestamp', Date.now()]
           ])
         } as const;
       } catch (error) {
         return {
           success: false,
           error: error.message,
           metadata: new Map([
             ['estimate', estimate],
             ['error', error]
           ])
         } as const;
       }
     };

   export const createGasOptimizer = (
     config: Readonly<GasConfig>
   ) =>
     async (
       message: Readonly<MessageHandler>
     ): Promise<Readonly<GasEstimate>> => {
       // Create estimator chain
       const estimators = [
         createGasEstimator(config),
         createHistoricalEstimator(),
         createNetworkEstimator()
       ];
       
       // Get estimates from all estimators
       const estimates = await Promise.all(
         estimators.map(e => e(message))
       );
       
       // Select optimal estimate
       return selectOptimalEstimate(estimates);
     };
   ```
